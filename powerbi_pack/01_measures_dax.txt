-- Create these measures in Power BI (Model view) using imported table names with spaces

Total Jobs = DISTINCTCOUNT('fact job_posting'[job_posting_key])

Jobs With Salary = CALCULATE([Total Jobs], 'fact job_posting'[has_salary_info] = TRUE())

Salary Coverage % = DIVIDE([Jobs With Salary], [Total Jobs])

Remote Jobs = CALCULATE([Total Jobs], 'fact job_posting'[is_work_from_home] = TRUE())

Remote Share % = DIVIDE([Remote Jobs], [Total Jobs])

Jobs No Degree = CALCULATE([Total Jobs], 'fact job_posting'[no_degree_mention] = TRUE())

No Degree % = DIVIDE([Jobs No Degree], [Total Jobs])

Jobs With Insurance = CALCULATE([Total Jobs], 'fact job_posting'[has_health_insurance] = TRUE())

Health Insurance % = DIVIDE([Jobs With Insurance], [Total Jobs])

Avg Salary Year = CALCULATE(AVERAGE('fact job_posting'[salary_year_avg]), 'fact job_posting'[has_salary_info] = TRUE())

Median Salary Year = CALCULATE(MEDIAN('fact job_posting'[salary_year_avg]), 'fact job_posting'[has_salary_info] = TRUE())

Remote Avg Salary Year = CALCULATE([Avg Salary Year], 'fact job_posting'[is_work_from_home] = TRUE())

Onsite Avg Salary Year = CALCULATE([Avg Salary Year], 'fact job_posting'[is_work_from_home] = FALSE())

Remote Salary Premium % = DIVIDE([Remote Avg Salary Year] - [Onsite Avg Salary Year], [Onsite Avg Salary Year])

Jobs PM = CALCULATE([Total Jobs], DATEADD('dim date'[full_date], -1, MONTH))

Jobs MoM % = DIVIDE([Total Jobs] - [Jobs PM], [Jobs PM])

Jobs PY = CALCULATE([Total Jobs], DATEADD('dim date'[full_date], -1, YEAR))

Jobs YoY % = DIVIDE([Total Jobs] - [Jobs PY], [Jobs PY])

Skill Mentions = COUNTROWS('bridge job_skill')

Distinct Skills = DISTINCTCOUNT('dim skill'[skill_key])

Jobs In Skill Context = DISTINCTCOUNT('bridge job_skill'[job_posting_key])

Avg Skills per Job = DIVIDE([Skill Mentions], [Jobs In Skill Context])

Schedule Mentions = COUNTROWS('bridge job_schedule')

Jobs In Schedule Context = DISTINCTCOUNT('bridge job_schedule'[job_posting_key])

Avg Schedule Types per Job = DIVIDE([Schedule Mentions], [Jobs In Schedule Context])

Top Portal Jobs = MAXX(ALL('dim job_portal'[portal_name]), CALCULATE([Total Jobs]))

Top Portal Share % = DIVIDE([Top Portal Jobs], [Total Jobs])

High Opportunity Jobs = CALCULATE([Total Jobs], 'fact job_posting'[has_salary_info] = TRUE(), 'fact job_posting'[is_work_from_home] = TRUE(), 'fact job_posting'[no_degree_mention] = TRUE())

High Opportunity Share % = DIVIDE([High Opportunity Jobs], [Total Jobs])

Opportunity Score = 0.30 * [Salary Coverage %] + 0.25 * [Remote Share %] + 0.25 * [Health Insurance %] + 0.20 * [No Degree %]
